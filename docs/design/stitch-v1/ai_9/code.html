import { ChainVigil } from '@chainvigil/sdk';

// Initialize with your API key
const client = new ChainVigil('cv_live_x89a...');

async function assessTradeRisk(targetAddress) {
  try {
    // Perform sync analysis before routing tx
    const report = await client.analyzeToken(targetAddress);
    
    if (report.riskScore > 85 || report.tags.includes('HONEYPOT')) {
      blockExecution(report.humanReadableSummary);
    } else {
      proceedWithTrade();
    }
  } catch (error) {
    handleError(error);
  }
}
